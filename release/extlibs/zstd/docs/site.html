API

接口用途
compress(data)压缩 Buffer 或 string，返回 Buffer；默认级别 3。
decompress(data)解压 Buffer，返回 Buffer；支持串联帧。
decompressText(data)将解压结果作为文本读取。
compressor() / decompressor()增量 update、drain、finish 和累计字节数。
compressFile() / decompressFile()文件处理、限额、覆盖选项和进度回调；含 NUL 的源/目标路径报 ValueError。

范例

import zstd
Buffer packed = zstd.compress("hello zstd")
assert zstd.decompressText(packed) == "hello zstd"

分段二进制与限额

import zstd

func main(...) as int
    auto encoder = zstd.compressor()
    encoder.update("one\0")
    Buffer packed = encoder.drain()
    encoder.update("two")
    assert encoder.finish()
    assert packed.append(encoder.drain())
    auto decoder = zstd.decompressor()
    decoder.update(packed)
    assert decoder.finish() and decoder.finished and decoder.members == 1
    assert decoder.drain() == Buffer.fromHex("6f6e650074776f")
    bool rejected = false
    try
        zstd.compress("one\0two", 3, 1024, 1)
    catch RangeError error
        rejected = true
    end
    assert rejected
    return 0
end

取消后保留旧目标，再次重试

import zstd

func main(...) as int
    string work = dir.temp(prefix = "xlang-zstd-example-")
    string source = path.join(work, "输入.txt")
    string target = path.join(work, "output.zst")
    string restored = path.join(work, "restored.txt")
    try
        assert file.writeText(source, "hello zstd")
        assert file.writeText(target, "keep")
        bool cancelled = false
        try
            zstd.compressFile(source, target, onProgress = func(int inputBytes,
                int totalBytes, int outputBytes) as bool
                return false
            end)
        catch CancelledError error
            cancelled = true
        end
        assert cancelled and file.readText(target) == "keep"
        zstd.compressFile(source, target)
        zstd.decompressFile(target, restored)
        assert file.readText(restored) == "hello zstd"
    finally
        if file.exists(source) then assert file.delete(source) end
        if file.exists(target) then assert file.delete(target) end
        if file.exists(restored) then assert file.delete(restored) end
        assert dir.delete(work)
    end
    return 0
end

包内 examples/basic.xl 与 examples/basic_c.c 同输入、同输出；C 范例可直接用 xl run extlibs/zstd/examples/basic_c.c 运行。C 调用方可直接包含 include/xrt_zstd.h，并负责释放返回的 xbuffer。C 流式接口的进度从 Reader 当前读取位置起算；无法同时取得总长和当前位置时不设置 TOTAL_KNOWN。C 编解码入口拒绝选项和资源限制结构的无效版本、长度及保留字段；无效选项不消费输入流。iMaxEntries 限制串联帧数，iMaxItemBytes 限制单帧原文大小；节点数、深度和未知标志不适用。资源限额、所有权及当前验收范围见包内 README。

包级C门禁既有XRT逐分配点检查，也有独立customMem后端分配失败夹具，不把前者当成后者的覆盖。原错误类型、中央用例和完整教学范例保留；短时门禁不替代32位、TSan或长时间压力验证。

独立C流式接口使用普通进度回调；正式native文件接口沿用XRT callable，见examples/file_transaction_c.c，不将两种ABI混称。C/XL完整事务范例还验证业务异常首因及恢复。普通取消例不代替双故障回滚；以上函数内例也不覆盖全局对象退出问题。官网只在隔离目录生成，不代表已部署。
