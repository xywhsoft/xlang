九、循环语句
================================================================================

9.1 for-in 遍历
---------------
// 遍历值
for v in iterable
    stmts
end

// 遍历索引和值
for i, v in iterable
    stmts
end

// 遍历字典键值
for k, v in dict
    stmts
end

// 仅遍历字典键
for k in dict
    stmts
end


9.2 for-in 遍历对象
-------------------
// 数组
for v in [1, 2, 3]
    stmts
end

// 列表
for i, v in myList
    stmts
end

// 集合
for v in mySet
    stmts
end

// 字典
for k, v in myDict
    stmts
end

// 字符串（遍历字符）
for ch in "hello"
    stmts
end

// 范围
for i in range(1, 10)
    stmts
end


9.3 range 循环
--------------
// 两参数（自动步长）
for i in range(1, 10)       // 1 到 10，步长 1
    stmts
end

for i in range(10, 1)       // 10 到 1，步长 -1
    stmts
end

// 三参数（指定步长）
for i in range(1, 10, 2)    // 1, 3, 5, 7, 9
    stmts
end

for i in range(10, 1, -2)   // 10, 8, 6, 4, 2
    stmts
end


9.4 while 循环（前置条件）
--------------------------
while expr do
    stmts
end

while a < 10 do
    a++
end


9.5 do-while 循环（后置条件）
-----------------------------
do
    stmts
while expr end

do
    a++
while a < 10 end


9.6 循环控制
------------
// break
for i in range(1, 100)
    if i > 10 then break
end

// break n（跳出 n 层）
for i in range(1, 10)
    for j in range(1, 10)
        if condition then break 2
    end
end

// continue
for i in range(1, 10)
    if i % 2 == 0 then continue
    print(i)
end

// continue n（跳过 n 层）
for i in range(1, 10)
    for j in range(1, 10)
        if condition then continue 2
    end
end


9.7 嵌套循环
------------
for i in range(1, 3)
    for j in range(1, 3)
        print(i, j)
    end
end

10.8 可变参数（数组形式）
-------------------------

    end

    for v in arg
        total += v
    end
    return total
end

    end
end

16.4 循环
---------
func main()
    // for-in 遍历数组
    array arr = [1, 2, 3, 4, 5]
    for i, v in arr
        println("arr[" + string(i) + "] = " + string(v))
    end
    
    // range 循环
    int sum = 0
    for i in range(1, 100)
        sum += i
    end
    println("1+2+...+100 = " + string(sum))
    
    // while 循环
    int n = 0
    while n < 5 do
        println("n = " + string(n))
        n++
    end
    
    // do-while 循环
    int m = 0
    do
        println("m = " + string(m))
        m++
    while m < 3 end
end


16.5 函数

    return a + b
end

func greet(string name, string msg = "Hello")
    println(msg + ", " + name + "!")
end

func swap(int a, int b) as int, int
    return b, a
end

func sum(...) as int
    int total = 0
    for v in arg
        total += v
    end
    return total
end

func main()
    println("add(3, 5) = " + string(add(3, 5)))
    
    greet("Tom")
    greet("Jerry", "Hi")
    
    auto (x, y) = swap(1, 2)
    println("swap: x=" + string(x) + ", y=" + string(y))
    
    println("sum(1,2,3,4,5) = " + string(sum(1, 2, 3, 4, 5)))
end

16.7 类型转换
-------------
func main()
    float f = 3.14
    int i = int(f)          // 截断为 3
    string s = string(i)    // 转为 "3"
    float f2 = float(i)     // 转为 3.0
    
    println("int(3.14) = " + string(i))
    println("string(3) = " + s)
    
    // 类型判断
    var v = 123
    if type(v) == int
        println("v is int")
    end
    
    v = "hello"
    if type(v) == string
        println("v is string")
    end
end
