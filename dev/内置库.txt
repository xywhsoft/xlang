


多线程（多线程以库的形式存在，不涉及语言特性，因此这一节废弃）：
	支持系统级多线程，利用多核 CPU 并行执行。
	
	创建线程：
		// 创建线程对象
		var t = thread(function(arg)
			print("Thread running: " + arg)
			return "result"
		end)
		
		// 启动线程
		t.start("hello")
		
		// 等待线程结束，获取返回值
		var result = t.join()
		
		// 线程属性
		t.id			// 线程ID
		t.status		// "pending" / "running" / "finished" / "error"
		t.detach()		// 分离线程（不需要 join）
	
	简化的 spawn 语法：
		// 启动并行任务（自动创建线程）
		var handle = spawn
			heavyComputation()
		end
		
		// 等待结果
		var result = await handle
	
	互斥锁：
		var mtx = mutex()
		
		mtx.lock()
		// 临界区代码
		mtx.unlock()
		
		// 自动解锁语法
		with mtx
			// 自动 lock/unlock
			sharedData += 1
		end
		
		// 尝试加锁（非阻塞）
		if mtx.trylock()
			// 成功获取锁
			mtx.unlock()
		end
	
	读写锁：
		var rwlock = rwmutex()
		
		rwlock.rlock()		// 读锁（共享）
		// 读取共享数据
		rwlock.runlock()
		
		rwlock.wlock()		// 写锁（独占）
		// 修改共享数据
		rwlock.wunlock()
	
	信号量：
		var sem = semaphore(5)	// 初始值 5
		
		sem.wait()		// P 操作，减 1
		sem.signal()	// V 操作，加 1
	
	条件变量：
		var cond = condition()
		var mtx = mutex()
		
		// 等待线程
		with mtx
			while not ready
				cond.wait(mtx)
			end
		end
		
		// 通知线程
		with mtx
			ready = true
			cond.signal()		// 唤醒一个
			cond.broadcast()	// 唤醒所有
		end
	
	原子操作：
		var counter = atomic(0)
		
		counter.load()			// 读取
		counter.store(10)		// 存储
		counter.add(1)			// 原子加
		counter.sub(1)			// 原子减
		counter.exchange(20)	// 交换
		counter.cas(old, new)	// 比较并交换
	
	通道 (Channel)：
		线程间通信的推荐方式。
		
		创建通道：
			var ch = channel()			// 无缓冲通道（同步）
			var ch = channel(10)		// 缓冲通道，容量 10
		
		发送和接收：
			ch.send(value)				// 发送（可能阻塞）
			var value = ch.recv()		// 接收（可能阻塞）
		
		非阻塞操作：
			var ok = ch.trysend(value)	// 尝试发送
			var value, ok = ch.tryrecv()	// 尝试接收
		
		关闭通道：
			ch.close()
		
		遍历通道（直到关闭）：
			for msg in ch
				print(msg)
			end
		
		select 多路复用：
			select
				case msg = ch1.recv()
					print("from ch1: " + msg)
				case msg = ch2.recv()
					print("from ch2: " + msg)
				case ch3.send("hello")
					print("sent to ch3")
				default
					print("no channel ready")
			end



线程池（线程池以库的形式存在，不涉及语言特性，因此这一节废弃）：
	管理一组工作线程，适合大量短任务。
	
	创建线程池：
		var pool = threadpool(4)	// 4 个工作线程
	
	提交任务：
		var future = pool.submit(function(x)
			return x * x
		end, 10)
		
		// 获取结果
		var result = future.get()	// 阻塞等待
		var result = await future	// 异步等待
	
	批量执行：
		var results = pool.map(function(x)
			return x * x
		end, [1, 2, 3, 4, 5])
	
	关闭线程池：
		pool.shutdown()			// 等待所有任务完成
		pool.terminate()		// 立即终止



协程与线程的配合（构思中）：
	// async/await 默认在单线程上运行（异步 I/O）
	async function fetchAll(urls)
		var results = []
		for url in urls
			var data = await http.get(url)	// 协程切换，不阻塞
			results.append(data)
		end
		return results
	end
	
	// 在线程池上执行 CPU 密集任务
	async function processData(data)
		var result = await threadpool.run(function()
			return heavyComputation(data)
		end)
		return result
	end
