import x.ui.auto

auto window = x.ui.auto.test_window()
assert window.valid() == true, "x.ui.auto helper window failed"

auto baseLocator = x.ui.auto.locator()
auto scopedBase = x.ui.auto.within(baseLocator, window.hwnd())
auto timedEdit = x.ui.auto.with_timeout(x.ui.auto.by_automation_id("1001", 50004), 2000)
auto editLocator = x.ui.auto.within(timedEdit, window.hwnd())
auto classLocator = x.ui.auto.within(x.ui.auto.by_class_name("Edit", "", 50004), window.hwnd())
auto nameLocator = x.ui.auto.within(x.ui.auto.by_name("Click", 50000), window.hwnd())
auto typeLocator = x.ui.auto.within(x.ui.auto.by_control_type(50004), window.hwnd())
auto missingProcessLocator = x.ui.auto.within(x.ui.auto.by_process_id(0), window.hwnd())
auto jsonLocator = x.ui.auto.within(x.ui.auto.locator_from_json('{"automationId":"1001","className":"Edit","name":"","controlType":50004,"processId":0}'), window.hwnd())

assert scopedBase.timeout_ms == 5000, "x.ui.auto locator default timeout failed"
assert editLocator.timeout_ms == 2000, "x.ui.auto with_timeout failed"
assert x.ui.auto.wait_exists(editLocator) == true, "x.ui.auto wait_exists by automation id failed"
assert x.ui.auto.wait_exists(classLocator) == true, "x.ui.auto wait_exists by class failed"
assert x.ui.auto.wait_exists(typeLocator) == true, "x.ui.auto wait_exists by control type failed"
assert x.ui.auto.wait_exists(jsonLocator) == true, "x.ui.auto locator_from_json failed"
assert x.ui.auto.wait_exists(missingProcessLocator) == false, "x.ui.auto missing process locator should not match"

assert x.ui.auto.type_text(editLocator, "auto-text") == true, "x.ui.auto type_text failed"
assert x.ui.auto.read(editLocator) == "auto-text", "x.ui.auto read failed"
assert x.ui.auto.click(nameLocator) == true, "x.ui.auto click failed"
assert string.contains(x.ui.auto.read(editLocator), "button-clicked"), "x.ui.auto click result failed"

assert x.ui.auto.screenshot(editLocator, "release/tmp/x_ui_auto_edit.png") == true, "x.ui.auto screenshot failed"
assert string.len(x.ui.auto.focused_locator_json()) > 0, "x.ui.auto focused_locator_json failed"

assert window.close() == true, "x.ui.auto window close failed"
assert window.release() == true, "x.ui.auto window release failed"
assert x.ui.auto.shutdown() == true, "x.ui.auto shutdown failed"

println("x.ui.auto extlib ok")
