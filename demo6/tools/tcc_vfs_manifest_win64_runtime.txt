# Runtime-only TCC builtin VFS manifest for demo6 Windows x64 xl.dll.
# Format:
#   dir|virtual-prefix|source-directory
#   file|virtual-name|source-file
#
# xl.dll only needs enough resources to compile and relocate generated C inside
# the host process. XRT/xruntime functions are provided by tcc_add_symbol;
# these compatibility paths therefore expose declaration-only SDKs. The runtime
# object and executable import libraries stay in the compiler-capable manifest.

dir|include|%XL6_EMBED_TCC_DIST_X64%\include
dir|include/__xl_runtime|%XL6_ROOT%\src\runtime|*.h

# The tracked TCC include directory owns the Win32/POSIX compatibility shims;
# the directory entry above embeds them without depending on release outputs.

file|include/xl.h|%XL6_ROOT%\release\include\xl.h
file|include/xl_sync_retirement.h|%XL6_ROOT%\release\include\xl_sync_retirement.h
file|include/xl_cancellation_retirement.h|%XL6_ROOT%\release\include\xl_cancellation_retirement.h
file|include/__xl_runtime/xl_runtime.h|%XL6_ROOT%\release\include\xl_runtime.h
file|include/__xl_runtime/xl_extlib_abi.h|%XL6_ROOT%\release\include\xl_extlib_abi.h
file|include/__xl_runtime/xruntime_decl.h|%XL6_ROOT%\release\include\xruntime_decl.h
file|lib/libtcc1.a|%XL6_EMBED_TCC_DIST_X64%\lib\libtcc1.a
# Keep native C source compatibility without embedding either implementation
# product. Generated xlang C already reaches xruntime_decl.h through xl_runtime.h.
file|lib/xrt/xrt.h|%XL6_ROOT%\release\include\xrt_decl.h
file|lib/xruntime/xruntime.h|%XL6_ROOT%\release\include\xruntime_decl.h
